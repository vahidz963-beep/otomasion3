-- ============================================================
-- ماژول تولید (کارخانه) - اسکیمای Supabase / PostgreSQL
-- فرض: جدول profiles(id uuid references auth.users, role text) از قبل وجود دارد
-- فرض: جدول‌های انبار به شکل warehouse_items / warehouse_transactions وجود دارند
--        (اگر نام واقعی فرق دارد، فقط بخش‌های علامت‌گذاری‌شده با ⚠️ را اصلاح کنید)
-- ============================================================

-- ------------------------------------------------------------
-- 1) ENUM ها
-- ------------------------------------------------------------
create type production_source_type as enum ('direct_order', 'rnd_project');

create type production_status as enum (
  'pending',            -- در انتظار شروع
  'planning',           -- در حال برنامه‌ریزی
  'in_progress',        -- در حال اجرا
  'quality_control',    -- در حال کنترل کیفیت
  'completed',          -- تکمیل‌شده
  'on_hold',            -- متوقف‌شده
  'cancelled',          -- لغوشده
  'rework'              -- نیاز به اصلاح دوباره
);

create type production_stage_name as enum (
  'planning', 'execution', 'quality_control', 'final_output'
);

create type stage_status as enum (
  'not_started', 'in_progress', 'completed', 'rejected', 'rework_needed'
);

-- ------------------------------------------------------------
-- 2) جدول اصلی سفارش تولید
-- ------------------------------------------------------------
create table production_orders (
  id uuid primary key default gen_random_uuid(),
  order_code text unique not null,              -- مثل PRD-1404-0001 (تولید در تریگر پایین)
  source_type production_source_type not null,
  source_id uuid not null,                      -- شناسه سفارش مستقیم یا پروژه R&D (بدون FK چون polymorphic)

  product_name_fa text not null,
  product_name_en text,
  description text,

  quantity_planned numeric not null check (quantity_planned > 0),
  quantity_produced numeric not null default 0,
  quantity_rejected numeric not null default 0,
  unit text not null default 'عدد',

  status production_status not null default 'pending',
  priority smallint not null default 3 check (priority between 1 and 5), -- 1 = بالاترین اولویت

  assigned_to uuid references profiles(id),     -- مسئول/اپراتور اصلی
  created_by uuid references profiles(id),

  planned_start_date date,
  planned_end_date date,
  actual_start_date timestamptz,
  actual_end_date timestamptz,

  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create index idx_production_orders_status on production_orders(status);
create index idx_production_orders_source on production_orders(source_type, source_id);

-- تولید خودکار order_code به فرم PRD-<سال شمسی تقریبی از سال میلادی>-XXXX
create sequence production_order_seq;

create or replace function generate_production_order_code()
returns trigger language plpgsql as $$
begin
  if new.order_code is null then
    new.order_code := 'PRD-' || to_char(now(), 'YYYY') || '-' ||
                       lpad(nextval('production_order_seq')::text, 4, '0');
  end if;
  return new;
end;
$$;

create trigger trg_production_order_code
before insert on production_orders
for each row execute function generate_production_order_code();

-- به‌روزرسانی خودکار updated_at
create or replace function set_updated_at()
returns trigger language plpgsql as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

create trigger trg_production_orders_updated_at
before update on production_orders
for each row execute function set_updated_at();

-- ------------------------------------------------------------
-- 3) مراحل تولید (۴ ردیف به‌ازای هر سفارش)
-- ------------------------------------------------------------
create table production_stage_logs (
  id uuid primary key default gen_random_uuid(),
  production_order_id uuid not null references production_orders(id) on delete cascade,
  stage production_stage_name not null,
  status stage_status not null default 'not_started',
  assigned_to uuid references profiles(id),
  started_at timestamptz,
  completed_at timestamptz,
  notes text,
  created_at timestamptz not null default now(),
  unique(production_order_id, stage)
);

-- ساخت خودکار ۴ مرحله هنگام ایجاد سفارش تولید
create or replace function create_default_stage_logs()
returns trigger language plpgsql as $$
begin
  insert into production_stage_logs (production_order_id, stage)
  values
    (new.id, 'planning'),
    (new.id, 'execution'),
    (new.id, 'quality_control'),
    (new.id, 'final_output');
  return new;
end;
$$;

create trigger trg_create_stage_logs
after insert on production_orders
for each row execute function create_default_stage_logs();

-- وقتی مرحله‌ای شروع/تمام می‌شود، وضعیت کلی سفارش را همگام کن
create or replace function sync_order_status_from_stage()
returns trigger language plpgsql as $$
declare
  v_order_id uuid := new.production_order_id;
begin
  if new.stage = 'planning' and new.status = 'in_progress' then
    update production_orders set status = 'planning', actual_start_date = coalesce(actual_start_date, now())
    where id = v_order_id;
  elsif new.stage = 'execution' and new.status = 'in_progress' then
    update production_orders set status = 'in_progress' where id = v_order_id;
  elsif new.stage = 'quality_control' and new.status = 'in_progress' then
    update production_orders set status = 'quality_control' where id = v_order_id;
  elsif new.stage = 'final_output' and new.status = 'completed' then
    update production_orders set status = 'completed', actual_end_date = now() where id = v_order_id;
  elsif new.status = 'rework_needed' then
    update production_orders set status = 'rework' where id = v_order_id;
  end if;
  return new;
end;
$$;

create trigger trg_sync_order_status
after update of status on production_stage_logs
for each row execute function sync_order_status_from_stage();

-- ------------------------------------------------------------
-- 4) مصرف مواد اولیه (ارتباط با انبار)
-- ------------------------------------------------------------
create table production_material_usage (
  id uuid primary key default gen_random_uuid(),
  production_order_id uuid not null references production_orders(id) on delete cascade,
  warehouse_item_id uuid,              -- ⚠️ در صورت اختلاف نام، به جدول واقعی انبار وصل کنید
  item_name text not null,             -- denormalized برای پایداری گزارش‌ها
  quantity_required numeric not null,
  quantity_consumed numeric not null default 0,
  unit text,
  warehouse_transaction_id uuid,       -- شناسه تراکنش خروج در انبار
  status text not null default 'pending' check (status in ('pending','partially_issued','issued')),
  created_at timestamptz not null default now()
);

create index idx_material_usage_order on production_material_usage(production_order_id);

-- تابع کسر مواد از انبار (⚠️ نام جدول/ستون‌های انبار را با ساختار واقعی خودتان تطبیق دهید)
create or replace function deduct_material_from_warehouse(p_usage_id uuid)
returns void language plpgsql as $$
declare
  v_usage production_material_usage%rowtype;
  v_tx_id uuid;
begin
  select * into v_usage from production_material_usage where id = p_usage_id;

  if v_usage.quantity_consumed >= v_usage.quantity_required then
    raise exception 'این مقدار قبلاً به‌طور کامل کسر شده است';
  end if;

  -- ⚠️ فرض: warehouse_items(id, current_stock) و warehouse_transactions(id, item_id, type, quantity, ref_table, ref_id)
  update warehouse_items
  set current_stock = current_stock - (v_usage.quantity_required - v_usage.quantity_consumed)
  where id = v_usage.warehouse_item_id;

  insert into warehouse_transactions (item_id, type, quantity, ref_table, ref_id)
  values (v_usage.warehouse_item_id, 'out', v_usage.quantity_required - v_usage.quantity_consumed,
          'production_material_usage', v_usage.id)
  returning id into v_tx_id;

  update production_material_usage
  set quantity_consumed = quantity_required,
      status = 'issued',
      warehouse_transaction_id = v_tx_id
  where id = p_usage_id;
end;
$$;

-- ------------------------------------------------------------
-- 5) کنترل کیفیت
-- ------------------------------------------------------------
create table production_quality_control (
  id uuid primary key default gen_random_uuid(),
  production_order_id uuid not null references production_orders(id) on delete cascade,
  checked_by uuid references profiles(id),
  check_date timestamptz not null default now(),
  quantity_checked numeric not null,
  quantity_approved numeric not null default 0,
  quantity_rejected numeric not null default 0,
  rejection_reason text,
  result text not null check (result in ('approved','rejected','partial')),
  notes text,
  created_at timestamptz not null default now()
);

-- به‌روزرسانی مقادیر تولید/رد در سفارش اصلی بعد از هر QC
create or replace function apply_qc_result()
returns trigger language plpgsql as $$
begin
  update production_orders
  set quantity_produced = quantity_produced + new.quantity_approved,
      quantity_rejected = quantity_rejected + new.quantity_rejected
  where id = new.production_order_id;
  return new;
end;
$$;

create trigger trg_apply_qc_result
after insert on production_quality_control
for each row execute function apply_qc_result();

-- ------------------------------------------------------------
-- 6) خروجی نهایی (ثبت در انبار)
-- ------------------------------------------------------------
create table production_output (
  id uuid primary key default gen_random_uuid(),
  production_order_id uuid not null references production_orders(id) on delete cascade,
  quantity numeric not null,
  warehouse_item_id uuid,             -- ⚠️ شناسه کالای نهایی در انبار
  warehouse_transaction_id uuid,
  delivered_at timestamptz not null default now(),
  registered_by uuid references profiles(id),
  created_at timestamptz not null default now()
);

create or replace function register_output_to_warehouse(
  p_production_order_id uuid,
  p_quantity numeric,
  p_warehouse_item_id uuid,
  p_registered_by uuid
) returns uuid language plpgsql as $$
declare
  v_tx_id uuid;
  v_output_id uuid;
begin
  -- ⚠️ فرض همان ساختار warehouse_items / warehouse_transactions بالا
  update warehouse_items
  set current_stock = current_stock + p_quantity
  where id = p_warehouse_item_id;

  insert into warehouse_transactions (item_id, type, quantity, ref_table, ref_id)
  values (p_warehouse_item_id, 'in', p_quantity, 'production_output', p_production_order_id)
  returning id into v_tx_id;

  insert into production_output (production_order_id, quantity, warehouse_item_id,
                                  warehouse_transaction_id, registered_by)
  values (p_production_order_id, p_quantity, p_warehouse_item_id, v_tx_id, p_registered_by)
  returning id into v_output_id;

  return v_output_id;
end;
$$;

-- ------------------------------------------------------------
-- 7) لاگ پیشرفت آزاد (برای گزارش به مدیریت)
-- ------------------------------------------------------------
create table production_progress_logs (
  id uuid primary key default gen_random_uuid(),
  production_order_id uuid not null references production_orders(id) on delete cascade,
  logged_by uuid references profiles(id),
  percent_complete smallint check (percent_complete between 0 and 100),
  note text,
  created_at timestamptz not null default now()
);

-- ------------------------------------------------------------
-- 8) ویو داشبورد مدیریتی
-- ------------------------------------------------------------
create or replace view production_dashboard_view as
select
  po.id,
  po.order_code,
  po.product_name_fa,
  po.product_name_en,
  po.source_type,
  po.status,
  po.priority,
  po.quantity_planned,
  po.quantity_produced,
  po.quantity_rejected,
  round(coalesce(po.quantity_produced,0) / nullif(po.quantity_planned,0) * 100, 1) as progress_percent,
  po.planned_start_date,
  po.planned_end_date,
  po.actual_start_date,
  po.actual_end_date,
  (po.planned_end_date < current_date and po.status not in ('completed','cancelled')) as is_delayed,
  po.assigned_to,
  (select count(*) filter (where sl.status = 'completed') from production_stage_logs sl
     where sl.production_order_id = po.id) as stages_completed,
  4 as stages_total
from production_orders po;

-- تابع کمکی برای گرفتن نقش کاربر جاری
create or replace function auth_role() returns text language sql stable as $$
  select role from profiles where id = auth.uid();
$$;

-- دسترسی کامل برای مدیر کل و مسئول تولید
create policy "full_access_production_orders" on production_orders
  for all using (auth_role() in ('مدیر_کل', 'مسئول_تولید'))
  with check (auth_role() in ('مدیر_کل', 'مسئول_تولید'));

-- فقط خواندن برای بقیه نقش‌ها
create policy "read_only_production_orders" on production_orders
  for select using (auth_role() in ('فروش', 'مسئول_rnd', 'انباردار', 'حسابدار', 'اداری'));
