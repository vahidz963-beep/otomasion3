-- B. Price calculator — item 5 of scope
--    Margin defaults per sales path, adjustable by admin; the calculator
--    itself is a SQL function combining a material cost (usually pulled from
--    the read-only warehouse view below) with the path's margin.
-- -----------------------------------------------------------------------------
CREATE TABLE price_margin_settings (
  sales_path        order_sales_path PRIMARY KEY,
  default_margin_pct numeric NOT NULL DEFAULT 20, -- percent
  updated_by        uuid REFERENCES profiles(id),
  updated_at        timestamptz NOT NULL DEFAULT now()
);

  USING (is_admin());

-- fn_calculate_price(material_cost, sales_path, override_margin_pct)
-- Returns suggested unit price = cost * (1 + margin/100). Pass override_margin_pct
-- to let the sales manager type a custom margin instead of the default.
CREATE OR REPLACE FUNCTION fn_calculate_price(
  p_material_cost numeric,
  p_sales_path order_sales_path,
  p_override_margin_pct numeric DEFAULT NULL
)
RETURNS numeric AS $$
DECLARE
  v_margin numeric;
BEGIN
  IF p_override_margin_pct IS NOT NULL THEN
    v_margin := p_override_margin_pct;
  ELSE
    SELECT default_margin_pct INTO v_margin
    FROM price_margin_settings WHERE sales_path = p_sales_path;
  END IF;

  RETURN round(p_material_cost * (1 + COALESCE(v_margin, 0) / 100.0), 2);
END;
$$ LANGUAGE plpgsql STABLE;

-- -----------------------------------------------------------------------------
-- C. Quotations (پیش‌فاکتور) — item 4 of scope
--    A quotation precedes a confirmed order. On approval it converts into a
--    real `orders` row (via fn_convert_quotation_to_order below); it never
--    mutates an existing order, keeping "one path per order" intact.
-- -----------------------------------------------------------------------------
CREATE TYPE quotation_status AS ENUM ('draft', 'sent', 'approved', 'rejected', 'expired', 'converted');

CREATE TABLE quotations (
  id                 uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  quotation_code     text UNIQUE NOT NULL,
  customer_id        uuid NOT NULL REFERENCES customers(id),
  sales_path         order_sales_path NOT NULL,
  status             quotation_status NOT NULL DEFAULT 'draft',
  title_fa           text NOT NULL,
  title_en           text,
  valid_until        date,
  discount_amount    numeric NOT NULL DEFAULT 0,
  tax_pct            numeric NOT NULL DEFAULT 0,
  sales_officer_id   uuid REFERENCES profiles(id),
  created_by         uuid NOT NULL REFERENCES profiles(id),
  converted_order_id uuid REFERENCES orders(id),
  notes              text,
  created_at         timestamptz NOT NULL DEFAULT now(),
  updated_at         timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX idx_quotations_customer ON quotations(customer_id);
CREATE INDEX idx_quotations_status ON quotations(status);

CREATE TABLE quotation_items (
  id               uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  quotation_id     uuid NOT NULL REFERENCES quotations(id) ON DELETE CASCADE,
  item_name_fa     text NOT NULL,
  item_name_en     text,
  warehouse_item_code text,
  quantity         numeric NOT NULL CHECK (quantity > 0),
  unit             text NOT NULL,
  material_cost    numeric NOT NULL DEFAULT 0, -- pulled from warehouse view at calc time
  margin_pct       numeric, -- NULL = use price_margin_settings default for the quotation's sales_path
  unit_price        numeric GENERATED ALWAYS AS (NULL) STORED, -- placeholder; real value set by fn_recalculate_quotation_item below (generated columns can't call functions)
  line_total       numeric,
  created_at       timestamptz NOT NULL DEFAULT now()
);

-- NOTE: unit_price as a plain STORED generated column can't call fn_calculate_price
-- (Postgres generated columns must be immutable-only expressions). Instead we
-- compute and store unit_price/line_total via trigger below, so the UI can
-- still just read the columns directly.
ALTER TABLE quotation_items DROP COLUMN unit_price;
ALTER TABLE quotation_items ADD COLUMN unit_price numeric;

CREATE OR REPLACE FUNCTION fn_recalculate_quotation_item()
RETURNS trigger AS $$
DECLARE
  v_path order_sales_path;
BEGIN
  SELECT sales_path INTO v_path FROM quotations WHERE id = NEW.quotation_id;
  NEW.unit_price := fn_calculate_price(NEW.material_cost, v_path, NEW.margin_pct);
  NEW.line_total := round(NEW.unit_price * NEW.quantity, 2);
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_recalculate_quotation_item
BEFORE INSERT OR UPDATE ON quotation_items
FOR EACH ROW EXECUTE FUNCTION fn_recalculate_quotation_item();

CREATE INDEX idx_quotation_items_quotation ON quotation_items(quotation_id);

-- Auto-generate quotation_code: QUO-YYYY-XXXX
CREATE SEQUENCE quotation_code_seq;

CREATE OR REPLACE FUNCTION fn_generate_quotation_code()
RETURNS trigger AS $$
DECLARE
  yr text := to_char(now(), 'YYYY');
BEGIN
  IF NEW.quotation_code IS NULL THEN
    NEW.quotation_code := 'QUO-' || yr || '-' || lpad(nextval('quotation_code_seq')::text, 4, '0');
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_generate_quotation_code
BEFORE INSERT ON quotations
FOR EACH ROW EXECUTE FUNCTION fn_generate_quotation_code();

-- Convert an approved quotation into a real order (idempotent: no-ops if
-- already converted). Copies items into order_items. Sales path/customer
-- carry over unchanged (a quotation and its resulting order always share
-- the same path, consistent with "one path per order").
CREATE OR REPLACE FUNCTION fn_convert_quotation_to_order(p_quotation_id uuid)
RETURNS uuid AS $$
DECLARE
  v_quotation quotations%ROWTYPE;
  v_order_id uuid;
BEGIN
  SELECT * INTO v_quotation FROM quotations WHERE id = p_quotation_id FOR UPDATE;

  IF v_quotation.status <> 'approved' THEN
    RAISE EXCEPTION 'Only approved quotations can be converted (quotation %)', v_quotation.quotation_code;
  END IF;

  IF v_quotation.converted_order_id IS NOT NULL THEN
    RETURN v_quotation.converted_order_id; -- already converted
  END IF;

  INSERT INTO orders (customer_id, sales_path, title_fa, title_en, sales_officer_id, created_by)
  VALUES (v_quotation.customer_id, v_quotation.sales_path, v_quotation.title_fa, v_quotation.title_en,
          v_quotation.sales_officer_id, v_quotation.created_by)
  RETURNING id INTO v_order_id;

  INSERT INTO order_items (order_id, item_name_fa, item_name_en, warehouse_item_code, quantity, unit, unit_price)
  SELECT v_order_id, item_name_fa, item_name_en, warehouse_item_code, quantity, unit, unit_price
  FROM quotation_items WHERE quotation_id = p_quotation_id;

  UPDATE quotations SET status = 'converted', converted_order_id = v_order_id, updated_at = now()
  WHERE id = p_quotation_id;

  RETURN v_order_id;
END;
$$ LANGUAGE plpgsql;

ALTER TABLE quotations ENABLE ROW LEVEL SECURITY;
ALTER TABLE quotation_items ENABLE ROW LEVEL SECURITY;

CREATE POLICY quotations_sales_manager ON quotations FOR ALL

CREATE POLICY quotation_items_via_quotation ON quotation_items FOR ALL
  USING (EXISTS (SELECT 1 FROM quotations q WHERE q.id = quotation_items.quotation_id));

-- Notify the target department the moment a request is created
CREATE OR REPLACE FUNCTION fn_notify_internal_request()
RETURNS trigger AS $$
DECLARE
  v_role user_role;
BEGIN
  v_role := CASE NEW.target_department
    WHEN 'warehouse' THEN 'warehouse_keeper'
    WHEN 'production' THEN 'production_officer'
    WHEN 'rnd' THEN 'rnd_officer'
  END;

  INSERT INTO notifications (recipient_role, title_fa, title_en, body_fa, related_order_id)
  VALUES (v_role, 'درخواست جدید از فروش: ' || NEW.subject_fa, 'New request from Sales', NEW.request_code, NEW.related_order_id);

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_notify_internal_request
AFTER INSERT ON internal_requests
FOR EACH ROW EXECUTE FUNCTION fn_notify_internal_request();

-- When the target department responds, notify the original requester back
CREATE OR REPLACE FUNCTION fn_notify_request_answered()
RETURNS trigger AS $$
BEGIN
  IF TG_OP = 'UPDATE' AND OLD.status <> 'answered' AND NEW.status = 'answered' THEN
    INSERT INTO notifications (recipient_id, title_fa, title_en, body_fa, related_order_id)
    VALUES (NEW.requested_by, 'پاسخ به درخواست ' || NEW.request_code, 'Your request was answered', NEW.response_fa, NEW.related_order_id);
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_notify_request_answered
BEFORE UPDATE ON internal_requests
FOR EACH ROW EXECUTE FUNCTION fn_notify_request_answered();

-- Auto-computed view: customers with no order in the last N days.
-- Reads system_settings.follow_up_threshold_days if that row exists,
-- otherwise defaults to 14. Adjust the table/column name once the Auth
-- session's system_settings shape is confirmed.
CREATE OR REPLACE VIEW v_customers_needing_followup AS
WITH threshold AS (
  SELECT COALESCE(
    (SELECT value::int FROM system_settings WHERE key = 'follow_up_threshold_days'),
    14
  ) AS days
),
last_activity AS (
  SELECT customer_id, MAX(created_at) AS last_order_at
  FROM orders
  GROUP BY customer_id
)
SELECT
  c.id AS customer_id,
  c.company_name,
  c.tier,
  la.last_order_at,
  (CURRENT_DATE - la.last_order_at::date) AS days_since_last_order
FROM customers c
LEFT JOIN last_activity la ON la.customer_id = c.id
CROSS JOIN threshold t
WHERE c.is_active
  AND (la.last_order_at IS NULL OR la.last_order_at < now() - (t.days || ' days')::interval);

-- -----------------------------------------------------------------------------
-- G. Personal sales performance report — item 8 of scope
--    Single RPC per the project's established dashboard pattern (DB-side
--    aggregation to minimize Supabase free-tier round-trips).
-- -----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION get_sales_performance_kpis(
  p_officer_id uuid,
  p_start_date date,
  p_end_date date
)
RETURNS json AS $$
DECLARE
  v_result json;
BEGIN
  SELECT json_build_object(
    'total_orders', COUNT(*) FILTER (WHERE o.sales_officer_id = p_officer_id),
    'orders_by_path', json_object_agg(o.sales_path, path_count) FILTER (WHERE o.sales_path IS NOT NULL),
    'closed_orders', COUNT(*) FILTER (WHERE o.sales_officer_id = p_officer_id AND o.current_stage = 'closed'),
    'cancelled_orders', COUNT(*) FILTER (WHERE o.sales_officer_id = p_officer_id AND o.is_cancelled),
    'quotations_sent', (SELECT COUNT(*) FROM quotations q WHERE q.sales_officer_id = p_officer_id AND q.status IN ('sent','approved','rejected','converted') AND q.created_at BETWEEN p_start_date AND p_end_date),
    'quotations_converted', (SELECT COUNT(*) FROM quotations q WHERE q.sales_officer_id = p_officer_id AND q.status = 'converted' AND q.created_at BETWEEN p_start_date AND p_end_date),
    'total_order_value', (SELECT COALESCE(SUM(oi.line_total), 0) FROM order_items oi JOIN orders o2 ON o2.id = oi.order_id WHERE o2.sales_officer_id = p_officer_id AND o2.created_at BETWEEN p_start_date AND p_end_date)
  ) INTO v_result
  FROM orders o
  LEFT JOIN (
    SELECT sales_path, COUNT(*) AS path_count FROM orders
    WHERE sales_officer_id = p_officer_id AND created_at BETWEEN p_start_date AND p_end_date
    GROUP BY sales_path
  ) path_counts ON true
  WHERE o.sales_officer_id = p_officer_id AND o.created_at BETWEEN p_start_date AND p_end_date;

  RETURN v_result;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;
