-- ---------------------------------------------------------------------
-- 1. ROLE ENUM (7 confirmed roles)
-- ---------------------------------------------------------------------
create type user_role as enum (
  'admin',        -- مدیر کل
  'sales',        -- فروش
  'rnd',          -- مسئول R&D
  'production',   -- مسئول تولید
  'warehouse',    -- انباردار
  'accountant',   -- حسابدار
  'office_admin'  -- اداری
);

-- ---------------------------------------------------------------------
-- 2. PROFILES TABLE (extends auth.users)
-- ---------------------------------------------------------------------
create table profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  full_name_fa text,
  full_name_en text,
  role user_role not null default 'sales',
  is_active boolean not null default true,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- ---------------------------------------------------------------------
-- 3. SYSTEM SETTINGS (global admin-configurable toggles)
--    Used for things like sales order visibility scope, without
--    requiring schema migrations later.
-- ---------------------------------------------------------------------
create table system_settings (
  key text primary key,
  value jsonb not null,
  updated_by uuid references profiles(id),
  updated_at timestamptz not null default now()
);

insert into system_settings (key, value) values
  ('sales_order_visibility', '"own"'::jsonb) -- 'own' | 'all'
on conflict (key) do nothing;

-- ---------------------------------------------------------------------
-- 4. SECURITY-DEFINER HELPER FUNCTIONS
--    Every module MUST use these three + any new ones added here.
-- ---------------------------------------------------------------------
create or replace function current_role_name() returns user_role
language sql security definer stable as $$
  select role from profiles where id = auth.uid();
$$;

create or replace function is_active_user() returns boolean
language sql security definer stable as $$
  select coalesce((select is_active from profiles where id = auth.uid()), false);
$$;

create or replace function is_admin() returns boolean
language sql security definer stable as $$
  select current_role_name() = 'admin' and is_active_user();
$$;

-- Sales visibility scope toggle (global, admin-configurable)
create or replace function sales_can_view_all_orders() returns boolean
language sql security definer stable as $$
  select coalesce(
    (select value = '"all"'::jsonb from system_settings where key = 'sales_order_visibility'),
    false
  );
$$;

-- ---------------------------------------------------------------------
-- 5. PRIVILEGE ESCALATION PREVENTION TRIGGER
--    Prevents a non-admin from changing their own role or active status.
-- ---------------------------------------------------------------------
create or replace function prevent_role_self_escalation() returns trigger
language plpgsql security definer as $$
begin
  if new.role <> old.role and not is_admin() then
    raise exception 'فقط مدیر کل می‌تواند نقش کاربران را تغییر دهد / Only admin can change roles';
  end if;
  if new.is_active <> old.is_active and not is_admin() then
    raise exception 'فقط مدیر کل می‌تواند وضعیت کاربر را تغییر دهد / Only admin can change active status';
  end if;
  return new;
end;
$$;

create trigger trg_prevent_role_escalation
before update on profiles
for each row execute function prevent_role_self_escalation();

-- ---------------------------------------------------------------------
-- 6. RLS: PROFILES
-- ---------------------------------------------------------------------
alter table profiles enable row level security;

create policy profiles_select_own_or_admin on profiles
for select using (id = auth.uid() or is_admin());

create policy profiles_update_own_or_admin on profiles
for update using (id = auth.uid() or is_admin());

create policy profiles_insert_admin_only on profiles
for insert with check (is_admin());

create policy profiles_delete_admin_only on profiles
for delete using (is_admin());

-- ---------------------------------------------------------------------
-- 7. RLS: SYSTEM_SETTINGS
-- ---------------------------------------------------------------------
alter table system_settings enable row level security;

create policy settings_select_active_users on system_settings
for select using (is_active_user());

create policy settings_update_admin_only on system_settings
for update using (is_admin());

create policy settings_insert_admin_only on system_settings
for insert with check (is_admin());
