-- -----------------------------------------------------------------------------
-- 3. Stage definitions per sales path — fully data-driven. There is NO fixed
--    stage count: each sales_path can have as many stages as the business
--    needs (2, 4, 12, whatever), added/removed/reordered by the admin at any
--    time via plain INSERT/UPDATE/DELETE on this table — no code or trigger
--    changes required. Two flags carry the behavior that used to be
--    hardcoded to specific stage names:
--      is_initial          -> exactly one "starting" stage per path (replaces
--                              the old hardcoded 'draft' check for content-lock)
--      notify_role_on_enter -> optional role to notify the moment an order
--                              enters this stage (replaces the old hardcoded
--                              per-stage-name IF/ELSE notification logic)
-- -----------------------------------------------------------------------------
CREATE TABLE order_status_definitions (
  id                   uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  sales_path           order_sales_path NOT NULL,
  stage_key            text NOT NULL,
  stage_order          numeric NOT NULL, -- numeric (not int) so a new stage can be slotted
                                          -- in between existing ones, e.g. 4.5, without renumbering
  stage_name_fa        text NOT NULL,
  stage_name_en        text NOT NULL,
  is_initial           boolean NOT NULL DEFAULT false,
  is_terminal          boolean NOT NULL DEFAULT false,
  notify_role_on_enter user_role, -- nullable; set to have a role notified automatically
  UNIQUE (sales_path, stage_key),
  UNIQUE (sales_path, stage_order)
);

-- Exactly one initial stage per path (the entry point orders are created into)
CREATE UNIQUE INDEX uq_order_status_one_initial_per_path
  ON order_status_definitions(sales_path) WHERE is_initial;

-- 4. Orders — master coordination record

CREATE TABLE orders (
  id                    uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  order_code            text UNIQUE NOT NULL,
  customer_id           uuid NOT NULL REFERENCES customers(id),
  sales_path            order_sales_path NOT NULL,

  cancelled_reason      text,

);

-- Reusable visibility check (also used by storage.objects policies below)
CREATE OR REPLACE FUNCTION fn_user_can_access_order(p_order_id uuid)
RETURNS boolean AS $$
  SELECT EXISTS (
    SELECT 1 FROM orders o
    WHERE o.id = p_order_id
      AND is_active_user()
      AND (
        is_admin()
        OR (current_role() = 'sales_manager' AND o.sales_officer_id = auth.uid())
        OR (current_role() = 'production_officer' AND o.sales_path = 'production')
        OR (current_role() = 'rnd_officer' AND o.sales_path = 'rnd')
        OR (current_role() = 'warehouse_keeper' AND o.current_stage IN ('warehouse_receipt','final_output','delivered'))
        OR (current_role() = 'accountant')
      )
  );
$$ LANGUAGE sql SECURITY DEFINER STABLE;

CREATE POLICY order_attachments_access ON order_attachments FOR ALL
  USING (fn_user_can_access_order(order_id));

-- 8. Auto-generate order_code: ORD-YYYY-XXXX
CREATE SEQUENCE order_code_seq;

CREATE OR REPLACE FUNCTION fn_generate_order_code()
RETURNS trigger AS $$
DECLARE
  yr text := to_char(now(), 'YYYY');
  seq_val int;
BEGIN
  IF NEW.order_code IS NULL THEN
    seq_val := nextval('order_code_seq');
    NEW.order_code := 'ORD-' || yr || '-' || lpad(seq_val::text, 4, '0');
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_generate_order_code
BEFORE INSERT ON orders
FOR EACH ROW EXECUTE FUNCTION fn_generate_order_code();

-- 9. Validate stage against order_status_definitions + auto-log transitions
CREATE OR REPLACE FUNCTION fn_validate_and_log_stage_change()
RETURNS trigger AS $$
DECLARE
  valid_stage boolean;
BEGIN
  -- On insert, if no stage was given, default to whichever stage is marked
  -- is_initial for this sales_path (works no matter what that stage is named
  -- or how many stages the path has).
  IF TG_OP = 'INSERT' AND NEW.current_stage IS NULL THEN
    SELECT stage_key INTO NEW.current_stage
    FROM order_status_definitions
    WHERE sales_path = NEW.sales_path AND is_initial;
  END IF;

  SELECT EXISTS (
    SELECT 1 FROM order_status_definitions
    WHERE sales_path = NEW.sales_path AND stage_key = NEW.current_stage
  ) INTO valid_stage;

  IF NOT valid_stage THEN
    RAISE EXCEPTION 'Invalid stage "%" for sales_path "%"', NEW.current_stage, NEW.sales_path;
  END IF;

  IF TG_OP = 'UPDATE' AND OLD.current_stage IS DISTINCT FROM NEW.current_stage THEN
    INSERT INTO order_status_history (order_id, from_stage, to_stage, changed_by)
    VALUES (NEW.id, OLD.current_stage, NEW.current_stage, NEW.sales_officer_id);
  END IF;

  NEW.updated_at := now();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_validate_stage
BEFORE INSERT OR UPDATE ON orders
FOR EACH ROW EXECUTE FUNCTION fn_validate_and_log_stage_change();

-- 10. Cross-module handoff: creates the linked detail record in the right
--     module the moment an order is created on that path.
--     NOTE: table/column names (title_fa, status vs stage) must be reconciled
--     against the actual rnd_projects / production_orders schemas.
CREATE OR REPLACE FUNCTION fn_order_path_handoff()
RETURNS trigger AS $$
BEGIN
  IF NEW.sales_path = 'rnd' AND NEW.rnd_project_id IS NULL THEN
    INSERT INTO rnd_projects (order_id, title_fa, title_en, status)
    VALUES (NEW.id, NEW.title_fa, NEW.title_en, 'new') -- map to actual initial state of the 9-state enum
    RETURNING id INTO NEW.rnd_project_id;
  END IF;

  IF NEW.sales_path = 'production' AND NEW.production_order_id IS NULL THEN
    INSERT INTO production_orders (order_id, title_fa, title_en, stage)
    VALUES (NEW.id, NEW.title_fa, NEW.title_en, 'planning')
    RETURNING id INTO NEW.production_order_id;
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_order_path_handoff
BEFORE INSERT ON orders
FOR EACH ROW EXECUTE FUNCTION fn_order_path_handoff();

-- 11a. Lock order content once it leaves the path's initial stage —
--      sales_manager can no longer edit title/description/path/customer
--      after the first stage transition, no matter how many stages the path
--      has or what they're named. Uses is_initial, not a hardcoded stage key.
--      Cancellation (is_cancelled/cancelled_reason) is intentionally NOT locked:
--      it needs no manager approval and can happen at any stage.
CREATE OR REPLACE FUNCTION fn_lock_order_content()
RETURNS trigger AS $$
DECLARE
  v_old_is_initial boolean;
BEGIN
  SELECT is_initial INTO v_old_is_initial
  FROM order_status_definitions
  WHERE sales_path = OLD.sales_path AND stage_key = OLD.current_stage;

  IF NOT COALESCE(v_old_is_initial, false) AND NOT is_admin() THEN
    IF NEW.title_fa IS DISTINCT FROM OLD.title_fa
       OR NEW.title_en IS DISTINCT FROM OLD.title_en
       OR NEW.description_fa IS DISTINCT FROM OLD.description_fa
       OR NEW.description_en IS DISTINCT FROM OLD.description_en
       OR NEW.sales_path IS DISTINCT FROM OLD.sales_path
       OR NEW.customer_id IS DISTINCT FROM OLD.customer_id THEN
      RAISE EXCEPTION 'Order content is locked once it leaves the initial stage (order %)', OLD.order_code;
    END IF;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_lock_order_content
BEFORE UPDATE ON orders
FOR EACH ROW EXECUTE FUNCTION fn_lock_order_content();

-- 11b. Lock order_items the same way (no add/edit/delete once past the initial stage)
CREATE OR REPLACE FUNCTION fn_lock_order_items()
RETURNS trigger AS $$
DECLARE
  v_is_initial boolean;
BEGIN
  SELECT d.is_initial INTO v_is_initial
  FROM orders o
  JOIN order_status_definitions d ON d.sales_path = o.sales_path AND d.stage_key = o.current_stage
  WHERE o.id = COALESCE(NEW.order_id, OLD.order_id);

  IF NOT COALESCE(v_is_initial, false) AND NOT is_admin() THEN
    RAISE EXCEPTION 'Order items are locked once the order leaves the initial stage';
  END IF;
  RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_lock_order_items
BEFORE INSERT OR UPDATE OR DELETE ON order_items
FOR EACH ROW EXECUTE FUNCTION fn_lock_order_items();

-- 11. Stage-change notifications — fully generic. Whichever role is set in
--     order_status_definitions.notify_role_on_enter for the NEW stage gets
--     notified. Adding a new stage with its own notification target is a
--     plain UPDATE on that table — no trigger code changes, ever.
CREATE OR REPLACE FUNCTION fn_notify_on_stage_change()
RETURNS trigger AS $$
DECLARE
  v_notify_role user_role;
  v_stage_name_fa text;
BEGIN
  IF TG_OP = 'UPDATE' AND OLD.current_stage IS DISTINCT FROM NEW.current_stage THEN
    SELECT notify_role_on_enter, stage_name_fa INTO v_notify_role, v_stage_name_fa
    FROM order_status_definitions
    WHERE sales_path = NEW.sales_path AND stage_key = NEW.current_stage;

    IF v_notify_role IS NOT NULL THEN
      INSERT INTO notifications (recipient_role, title_fa, title_en, body_fa, related_order_id)
      VALUES (v_notify_role, 'سفارش ' || NEW.order_code || ' وارد مرحله «' || v_stage_name_fa || '» شد',
              'Order ' || NEW.order_code || ' entered stage: ' || v_stage_name_fa, NEW.order_code, NEW.id);
    END IF;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_notify_stage_change
AFTER UPDATE ON orders
FOR EACH ROW EXECUTE FUNCTION fn_notify_on_stage_change();

-- Sales Manager (مدیر فروش): see all orders (B2B cross-visibility), insert their own,
-- update/cancel only their own. No separate manager-approval gate for cancellation.
CREATE POLICY orders_sales_select ON orders FOR SELECT
  USING (is_active_user() AND current_role() IN ('sales_manager','general_manager'));

CREATE POLICY orders_sales_insert ON orders FOR INSERT
  WITH CHECK (is_active_user() AND current_role() = 'sales_manager' AND created_by = auth.uid());

CREATE POLICY orders_sales_update ON orders FOR UPDATE
  USING (is_active_user() AND current_role() = 'sales_manager' AND sales_officer_id = auth.uid());

-- R&D officer: only rnd-path orders
CREATE POLICY orders_rnd_select ON orders FOR SELECT
  USING (is_active_user() AND current_role() = 'rnd_officer' AND sales_path = 'rnd');

-- Production officer: only production-path orders
CREATE POLICY orders_production_select ON orders FOR SELECT
  USING (is_active_user() AND current_role() = 'production_officer' AND sales_path = 'production');

-- Warehouse keeper: only orders currently at a warehouse-relevant stage
CREATE POLICY orders_warehouse_select ON orders FOR SELECT
  USING (is_active_user() AND current_role() = 'warehouse_keeper'
         AND current_stage IN ('warehouse_receipt','final_output','delivered'));

-- Accountant: read access to all orders (for invoicing/settlement context)
CREATE POLICY orders_accountant_select ON orders FOR SELECT
  USING (is_active_user() AND current_role() = 'accountant');

-- order_items / order_status_history: readable by anyone who can read the parent order
CREATE POLICY order_items_via_order ON order_items FOR SELECT
  USING (EXISTS (SELECT 1 FROM orders o WHERE o.id = order_items.order_id));

CREATE POLICY order_history_via_order ON order_status_history FOR SELECT
  USING (EXISTS (SELECT 1 FROM orders o WHERE o.id = order_status_history.order_id));

-- Notifications: visible to the named recipient, their role, or admin
CREATE POLICY notifications_own ON notifications FOR SELECT
  USING (recipient_id = auth.uid() OR recipient_role = current_role() OR is_admin());

CREATE POLICY notifications_mark_read ON notifications FOR UPDATE
  USING (recipient_id = auth.uid() OR recipient_role = current_role());

-- =============================================================================
-- 12. Unified tracking view (for dashboards + internal order lookup)
-- =============================================================================
CREATE VIEW v_order_tracking AS
SELECT
  o.id,
  o.order_code,
  c.company_name        AS customer_name,
  c.tier                AS customer_tier,
  o.sales_path,
  o.current_stage,
  d.stage_name_fa,
  d.stage_name_en,
  d.stage_order,
  d.is_terminal,
  o.priority,
  o.expected_delivery_date,
  o.is_cancelled,
  o.created_at,
  o.updated_at,
  p.full_name           AS sales_officer_name
FROM orders o
JOIN customers c               ON c.id = o.customer_id
JOIN order_status_definitions d ON d.sales_path = o.sales_path AND d.stage_key = o.current_stage
LEFT JOIN profiles p            ON p.id = o.sales_officer_id;
